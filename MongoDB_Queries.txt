2. GREATER THAN ($gt)

db.students.find({ marks: { $gt: 80 } })


3. LESS THAN ($lt)

db.students.find({ marks: { $lt: 80 } })


4. EQUAL TO ($eq)

db.students.find({ marks: { $eq: 85 } })


5. GREATER THAN OR EQUAL TO ($gte)

db.students.find({ marks: { $gte: 80 } })


6. NOT EQUAL TO ($ne)

db.students.find({ marks: { $ne: 85 } })


7. LESS THAN OR EQUAL TO ($lte)

db.students.find({ marks: { $lte: 80 } })


8. AND OPERATOR ($and)

db.students.find({
  $and: [
    { marks: { $gt: 70 } },
    { age: { $lt: 23 } }
  ]
})


9. OR OPERATOR ($or)

db.students.find({
  $or: [
    { department: "CSE" },
    { department: "ECE" }
  ]
})


10. NOT OPERATOR ($not)

db.students.find({
  marks: { $not: { $gt: 80 } }
})


11. UPDATE ONE

db.students.updateOne(
  { name: "Kunal" },
  { $set: { marks: 90 } }
)


12. SORT - ASCENDING

db.students.find().sort({ marks: 1 })


13. SORT - DESCENDING

db.students.find().sort({ marks: -1 })


14. LIMIT

db.students.find().sort({ marks: -1 }).limit(5)


15. SEARCH BY DEPARTMENT

db.students.find({ department: "CSE" })


16. DELETE ONE

db.students.deleteOne({ name: "Kunal" })


17. DELETE MANY

db.students.deleteMany({
  marks: { $lt: 65 }
})
